```typescript
// 替换原有的 edge:added 监听器
graph.on('edge:connected', ({ edge, isNew }: { edge: Edge; isNew: boolean }) => {
  if (!isNew) return
  
  const source = edge.getSourceCellId()
  const target = edge.getTargetCellId()
  
  // 1. 禁止自连
  if (source === target) {
    edge.remove()
    ElMessage.warning('不能连接同一个节点')
    return
  }
  
  // 2. 禁止重复连线
  const existingEdge = graph!.getEdges().find(e => 
    e.getSourceCellId() === source && e.getTargetCellId() === target
  )
  if (existingEdge && existingEdge !== edge) {
    edge.remove()
    ElMessage.warning('连线已存在')
    return
  }
  
  // 3. 条件节点校验（必须连出两条线）
  const sourceNode = store.flowDefinition.nodes.find(n => n.nodeId === source)
  if (sourceNode?.type === 'condition') {
    const edges = graph!.getEdges().filter(e => e.getSourceCellId() === source)
    if (edges.length > 2) {
      edge.remove()
      ElMessage.warning('条件节点最多只能有两条连线（true/false）')
      return
    }
    
    // 自动设置分支标签
    const existingTrue = edges.find(e => e.getLabel() === 'true')
    const existingFalse = edges.find(e => e.getLabel() === 'false')
    
    if (!existingTrue && !existingFalse) {
      edge.setLabelAt(0, 'true')
      ElMessage.info('连线已标记为 true 分支，可双击标签修改')
    } else if (!existingTrue) {
      edge.setLabelAt(0, 'true')
    } else if (!existingFalse) {
      edge.setLabelAt(0, 'false')
    }
  }
  
  // 4. 存储到 store
  const label = edge.getLabelAt(0)?.attrs?.label?.text || ''
  store.addEdge({ source, target, label })
})
```

```typescript
graph.on('history:change', ({ cmd, options }) => {
  // 将 X6 的撤销/重做与 store 同步
  if (cmd === 'undo') {
    syncGraphToStore()
  } else if (cmd === 'redo') {
    syncGraphToStore()
  }

// 同步函数
function syncGraphToStore() {
  if (!graph) return
  
  const cells = graph.getCells()
  const nodes = cells
    .filter(cell => cell.isNode())
    .map(node => ({
      nodeId: node.id,
      type: node.data.type,
      label: node.attr('label/text'),
      x: node.getPosition().x,
      y: node.getPosition().y,
      ...node.data,
    }))
  
  const edges = cells
    .filter(cell => cell.isEdge())
    .map(edge => ({
      source: edge.getSourceCellId(),
      target: edge.getTargetCellId(),
      label: edge.getLabelAt(0)?.attrs?.label?.text || '',
    }))
  
  store.setFlowDefinition({ nodes, edges })
}

// 工具栏撤销/重做调用 X6 的 API
function handleUndo() {
  graph?.undo()
  syncGraphToStore()
}

function handleRedo() {
  graph?.redo()
  syncGraphToStore()
}
```

```typescript
// 添加预览模拟执行
async function simulateExecution(inputData: any) {
  if (!graph) return []
  
  const startNode = store.flowDefinition.nodes.find(n => n.type === 'start')
  if (!startNode) return []
  
  const path = [startNode.nodeId]
  let currentNode = startNode
  
  while (currentNode.type !== 'end') {
    const edges = graph.getEdges().filter(e => e.getSourceCellId() === currentNode.nodeId)
    
    if (edges.length === 0) break
    
    let nextNodeId: string | null = null
    
    if (currentNode.type === 'condition') {
      // 执行表达式
      const expression = currentNode.expression || 'true'
      const result = evaluateExpression(expression, inputData)
      
      const edge = edges.find(e => e.getLabelAt(0)?.attrs?.label?.text === String(result))
      nextNodeId = edge?.getTargetCellId() || null
    } else {
      // 非条件节点，取第一条连线
      nextNodeId = edges[0]?.getTargetCellId()
    }
    
    if (!nextNodeId) break
    
    path.push(nextNodeId)
    currentNode = store.flowDefinition.nodes.find(n => n.nodeId === nextNodeId)!
  }
  
  return path
}

// 简单表达式求值（可用 expr-eval 或 jexl 库）
function evaluateExpression(expr: string, data: any): boolean {
  try {
    // 替换 fact.xxx 为 data.xxx
    const jsExpr = expr.replace(/fact\.(\w+)/g, (_, key) => {
      const value = data[key]
      return typeof value === 'string' ? `'${value}'` : String(value)
    })
    return new Function(`return ${jsExpr}`)()
  } catch {
    return false
  }
}

// 修改预览按钮逻辑
async function handlePreview() {
  if (store.isPreviewMode) {
    // 退出预览
    store.setPreviewMode(false)
    renderGraph(store.flowDefinition) // 重新渲染清除高亮
    return
  }
  
  // 进入预览，弹出测试数据输入框
  const { value: inputData } = await ElMessageBox.prompt('请输入测试数据 (JSON)', '预览模式', {
    confirmButtonText: '执行',
    cancelButtonText: '取消',
    inputValue: '{"patientType": "RESIDENT", "age": 30}',
    inputType: 'textarea',
  })
  
  try {
    const data = JSON.parse(inputData)
    const path = await simulateExecution(data)
    store.setPreviewPath(path)
    store.setPreviewMode(true)
    
    // 高亮路径
    renderGraph(store.flowDefinition)
    
    ElMessage.success(`执行路径: ${path.join(' → ')}`)
  } catch (error) {
    ElMessage.error('JSON 格式错误或执行失败')
  }
}
```

```vue

<script setup>
const showErrorPanel = ref(false)
const errors = ref<Array<{ type: 'error' | 'warning', message: string, nodeId?: string }>>([])

// 实时校验函数
function validateFlow() {
  errors.value = []
  
  const nodes = store.flowDefinition.nodes
  const edges = store.flowDefinition.edges
  
  // 1. 必须有开始和结束节点
  if (!nodes.find(n => n.type === 'start')) {
    errors.value.push({ type: 'error', message: '缺少开始节点' })
  }
  if (!nodes.find(n => n.type === 'end')) {
    errors.value.push({ type: 'error', message: '缺少结束节点' })
  }
  
  // 2. 孤立节点检查
  const connectedNodeIds = new Set([
    ...edges.map(e => e.source),
    ...edges.map(e => e.target),
  ])
  nodes.forEach(node => {
    if (node.type !== 'start' && !connectedNodeIds.has(node.nodeId)) {
      errors.value.push({ 
        type: 'warning', 
        message: `节点 "${node.label}" 是孤立的`,
        nodeId: node.nodeId,
      })
    }
  })
  
  // 3. 条件节点必须有两个分支
  nodes.forEach(node => {
    if (node.type === 'condition') {
      const outgoingEdges = edges.filter(e => e.source === node.nodeId)
      if (outgoingEdges.length !== 2) {
        errors.value.push({
          type: 'error',
          message: `条件节点 "${node.label}" 必须有两条连线（true/false）`,
          nodeId: node.nodeId,
        })
      }
    }
  })
  
  // 4. 检查是否有未连线的端口
  nodes.forEach(node => {
    const incomingEdges = edges.filter(e => e.target === node.nodeId)
    const outgoingEdges = edges.filter(e => e.source === node.nodeId)
    
    if (node.type !== 'start' && incomingEdges.length === 0) {
      errors.value.push({
        type: 'warning',
        message: `节点 "${node.label}" 没有输入连线`,
        nodeId: node.nodeId,
      })
    }
    if (node.type !== 'end' && outgoingEdges.length === 0) {
      errors.value.push({
        type: 'warning',
        message: `节点 "${node.label}" 没有输出连线`,
        nodeId: node.nodeId,
      })
    }
  })
  
  // 5. 检测循环依赖
  if (hasCycle(nodes, edges)) {
    errors.value.push({ type: 'error', message: '存在循环依赖，无法执行' })
  }
}

// 循环检测算法
function hasCycle(nodes: NodeDTO[], edges: EdgeDTO[]): boolean {
  const graph = new Map<string, string[]>()
  nodes.forEach(node => graph.set(node.nodeId, []))
  edges.forEach(edge => graph.get(edge.source)?.push(edge.target))
  
  const visited = new Set<string>()
  const recursionStack = new Set<string>()
  
  function dfs(nodeId: string): boolean {
    visited.add(nodeId)
    recursionStack.add(nodeId)
    
    for (const neighbor of graph.get(nodeId) || []) {
      if (!visited.has(neighbor)) {
        if (dfs(neighbor)) return true
      } else if (recursionStack.has(neighbor)) {
        return true
      }
    }
    
    recursionStack.delete(nodeId)
    return false
  }
  
  for (const node of nodes) {
    if (!visited.has(node.nodeId)) {
      if (dfs(node.nodeId)) return true
    }
  }
  return false
}

// 定位错误节点
function locateError(nodeId: string) {
  const node = graph?.getCell(nodeId)
  if (node && graph) {
    graph.centerCell(node)
    graph.select(node)
  }
}

// 监听变化自动校验
watch(() => store.flowDefinition, () => {
  validateFlow()
}, { deep: true })
</script>
```

```typescript
// 批量对齐实现
function batchAlign(direction: string) {
  const nodes = store.multiSelectedIds
    .map(id => graph?.getCell(id) as Node)
    .filter(n => n && n.isNode())
  
  if (nodes.length < 2) return
  
  if (direction === 'left') {
    const minX = Math.min(...nodes.map(n => n.getPosition().x))
    nodes.forEach(n => n.setPosition(minX, n.getPosition().y))
  } else if (direction === 'center') {
    const centerY = nodes.reduce((sum, n) => sum + n.getPosition().y, 0) / nodes.length
    nodes.forEach(n => n.setPosition(n.getPosition().x, centerY))
  } else if (direction === 'top') {
    const minY = Math.min(...nodes.map(n => n.getPosition().y))
    nodes.forEach(n => n.setPosition(n.getPosition().x, minY))
  }
  
  syncGraphToStore()
}
```

```typescript
import { DagreLayout } from '@antv/layout'
function autoLayout() {
  const nodes = store.flowDefinition.nodes.map(n => ({
    id: n.nodeId,
    data: n,
  }))
  const edges = store.flowDefinition.edges
  
  const layout = new DagreLayout({
    type: 'dagre',
    rankdir: 'TB',  // 从上到下排列
    align: 'UL',
    nodesep: 50,
    ranksep: 80,
  })
  
  const layoutResult = layout.layout({ nodes, edges })
  layoutResult.nodes.forEach(node => {
    store.updateNode(node.id, { x: node.x, y: node.y })
  })
  renderGraph(store.flowDefinition)
}
```
